cmake_minimum_required(VERSION 3.22.1)
project(paperforge_scan LANGUAGES C CXX ASM)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_QUANTOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
set(MNN_VULKAN OFF CACHE BOOL "" FORCE)
set(MNN_ARM82 OFF CACHE BOOL "" FORCE)
set(MNN_SUPPORT_BF16 OFF CACHE BOOL "" FORCE)
set(MNN_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(MNN_SME2 OFF CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
if(PAPERFORGE_MNN_SOURCE)
    add_subdirectory("${PAPERFORGE_MNN_SOURCE}" mnn)
else()
    include(FetchContent)
    FetchContent_Declare(mnn
        URL https://codeload.github.com/alibaba/MNN/zip/ea44a3ebd5dd6348eea501047b17c43aa3ecccb6
        URL_HASH SHA256=aa744c452e2b1b39e9442b1862051a02bdf3481193d26c9c1acb1380dba38ec8
    )
    FetchContent_MakeAvailable(mnn)
endif()
add_library(paperforge_scan SHARED scan_runtime.cpp)
target_include_directories(paperforge_scan PRIVATE "${mnn_SOURCE_DIR}/include")
if(PAPERFORGE_MNN_SOURCE)
    target_include_directories(paperforge_scan PRIVATE "${PAPERFORGE_MNN_SOURCE}/include")
endif()
target_link_libraries(paperforge_scan PRIVATE MNN log android)
target_link_options(paperforge_scan PRIVATE "-Wl,-z,max-page-size=16384")
